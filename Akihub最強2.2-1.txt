local function LocalHRP()

local function attachPhysics(p)
    if not p or not p.Parent then return nil, nil end
    pcall(function() p:SetNetworkOwner(LP) end)

    p.CanCollide = false
    p.CanTouch = false
    p.CanQuery = false
    p.Massless = true

    p.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    p.AssemblyAngularVelocity = Vector3.new(0, 0, 0)

    local bp = p:FindFirstChild("BodyPosition") or Instance.new("BodyPosition")
    if not bp.Parent then
        bp.Name = "BodyPosition"
        bp.MaxForce = Vector3.new(math.huge, math.huge, math.huge)
        bp.P = 3e6
        bp.D = 2000
        bp.Position = p.Position
        bp.Parent = p
    end
    local bg = p:FindFirstChild("BodyGyro") or Instance.new("BodyGyro")
    if not bg.Parent then
        bg.Name = "BodyGyro"
        bg.MaxTorque = Vector3.new(math.huge, math.huge, math.huge)
        bg.P = 3e6
        bg.D = 1000
        bg.CFrame = p.CFrame
        bg.Parent = p
    end

    return bp, bg
end

local function detachPhysics(p)
    if not p or not p.Parent then return end
    local bp = p:FindFirstChild("BodyPosition"); if bp then bp:Destroy() end
    local bg = p:FindFirstChild("BodyGyro"); if bg then bg:Destroy() end

    p.CanCollide = true
    p.CanTouch = true
    p.CanQuery = true
    p.Massless = false
    pcall(function() p:SetNetworkOwner(nil) end)
end

local function GetClosestPlayerInRange()
    local closestPlayer = nil
    local closestDistance = SilentAimRange
    local localRoot = LocalHRP()
    
    if not localRoot then return nil end
    
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= LP and player.Character then
            local character = player.Character
            local humanoid = character:FindFirstChild("Humanoid")
            local head = character:FindFirstChild("Head")
            
            if humanoid and humanoid.Health > 0 and head then
                local distance = (localRoot.Position - head.Position).Magnitude
                if distance < closestDistance then
                    closestDistance = distance
                    closestPlayer = head
                end
            end
        end
    end
    
    return closestPlayer
end

local function HookSilentAim()
    if oldNamecall then return end
    
    oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
        local args = {...}
        local method = getnamecallmethod()
        
        if SilentAimEnabled and tostring(method) == "Raycast" and self == workspace then
            local target = GetClosestPlayerInRange()
            if target then
                -- ターゲットの位置に向かう方向を計算
                local origin = args[1] or Vector3.new()
                local direction = (target.Position - origin).Unit
                args[3] = direction
            end
        end
        
        return oldNamecall(self, unpack(args))
    end)
end

    b.Position = targetCFrame.Position
    b.P, b.D = 2e4, 5e3

end

local function getPatternTarget(i, n, t)
    local root = LocalHRP(); if not root then return Vector3.new(0,0,0) end
    local phase = t * (RotationSpeed / 100) + (i * 2 * math.pi / n)
    local x, z, y = 0, 0, HeightOffset

    if CurrentPattern == "Circle" then
        x = Radius * math.cos(phase)
        z = Radius * math.sin(phase)
        
    elseif CurrentPattern == "Square" then
        local normalizedPhase = (phase % (2 * math.pi))
        local side = math.floor(normalizedPhase / (math.pi / 2))
        local progress = (normalizedPhase % (math.pi / 2)) / (math.pi / 2)
        
        if side == 0 then
            x = Radius
            z = -Radius + (2 * Radius * progress)
        elseif side == 1 then
            x = Radius - (2 * Radius * progress)
            z = Radius
        elseif side == 2 then
            x = -Radius
            z = Radius - (2 * Radius * progress)
        else
            x = -Radius + (2 * Radius * progress)
            z = -Radius
        end
        
    elseif CurrentPattern == "Triangle" then
        local normalizedPhase = (phase % (2 * math.pi))
        local side = math.floor(normalizedPhase / (2 * math.pi / 3))
        local progress = (normalizedPhase % (2 * math.pi / 3)) / (2 * math.pi / 3)
        
        local angles = {math.pi / 2, math.pi / 2 + 2 * math.pi / 3, math.pi / 2 + 4 * math.pi / 3}
        local x1 = Radius * math.cos(angles[side + 1])
        local z1 = Radius * math.sin(angles[side + 1])
        local x2 = Radius * math.cos(angles[(side + 1) % 3 + 1])
        local z2 = Radius * math.sin(angles[(side + 1) % 3 + 1])
        
        x = x1 + (x2 - x1) * progress
        z = z1 + (z2 - z1) * progress
        
    elseif CurrentPattern == "Pentagon" then
        local normalizedPhase = (phase % (2 * math.pi))
        local side = math.floor(normalizedPhase / (2 * math.pi / 5))
        local progress = (normalizedPhase % (2 * math.pi / 5)) / (2 * math.pi / 5)
        
        local angle1 = (side / 5) * 2 * math.pi - math.pi / 2
        local angle2 = ((side + 1) / 5) * 2 * math.pi - math.pi / 2
        
        local x1 = Radius * math.cos(angle1)
        local z1 = Radius * math.sin(angle1)
        local x2 = Radius * math.cos(angle2)
        local z2 = Radius * math.sin(angle2)
        
        x = x1 + (x2 - x1) * progress
        z = z1 + (z2 - z1) * progress
        
    elseif CurrentPattern == "Hexagon" then
        local normalizedPhase = (phase % (2 * math.pi))
        local side = math.floor(normalizedPhase / (2 * math.pi / 6))
        local progress = (normalizedPhase % (2 * math.pi / 6)) / (2 * math.pi / 6)
        
        local angle1 = (side / 6) * 2 * math.pi
        local angle2 = ((side + 1) / 6) * 2 * math.pi
        
        local x1 = Radius * math.cos(angle1)
        local z1 = Radius * math.sin(angle1)
        local x2 = Radius * math.cos(angle2)
        local z2 = Radius * math.sin(angle2)
        
        x = x1 + (x2 - x1) * progress
        z = z1 + (z2 - z1) * progress
        
    elseif CurrentPattern == "Octagon" then
        local normalizedPhase = (phase % (2 * math.pi))
        local side = math.floor(normalizedPhase / (2 * math.pi / 8))
        local progress = (normalizedPhase % (2 * math.pi / 8)) / (2 * math.pi / 8)
        
        local angle1 = (side / 8) * 2 * math.pi
        local angle2 = ((side + 1) / 8) * 2 * math.pi
        
        local x1 = Radius * math.cos(angle1)
        local z1 = Radius * math.sin(angle1)
        local x2 = Radius * math.cos(angle2)
        local z2 = Radius * math.sin(angle2)
        
        x = x1 + (x2 - x1) * progress
        z = z1 + (z2 - z1) * progress
        
    elseif CurrentPattern == "Star" then
        local outerRadius = Radius
        local innerRadius = Radius * 0.5
        local points = 5
        
        local segment = phase % (2 * math.pi / points)
        local point = math.floor(phase / (2 * math.pi / points))
        
        if point % 2 == 0 then
            local startAngle = point * (2 * math.pi / points)
            local endAngle = (point + 1) * (2 * math.pi / points)
            local progress = segment / (2 * math.pi / points)
            
            x = math.cos(startAngle + (endAngle - startAngle) * progress) * outerRadius
            z = math.sin(startAngle + (endAngle - startAngle) * progress) * outerRadius
        else
            local startAngle = point * (2 * math.pi / points)
            local endAngle = (point + 1) * (2 * math.pi / points)
            local progress = segment / (2 * math.pi / points)
            
            x = math.cos(startAngle + (endAngle - startAngle) * progress) * innerRadius
            z = math.sin(startAngle + (endAngle - startAngle) * progress) * innerRadius
        end
        
    elseif CurrentPattern == "Heart" then
        local heartPhase = phase * 2
        x = Radius * 16 * math.pow(math.sin(heartPhase), 3)
        z = Radius * (13 * math.cos(heartPhase) - 5 * math.cos(2 * heartPhase) - 2 * math.cos(3 * heartPhase) - math.cos(4 * heartPhase))
        x = x * 0.1
        z = z * 0.1
        
    elseif CurrentPattern == "Spiral" then
        local spiralFactor = 0.5
        x = Radius * math.cos(phase) * math.exp(-spiralFactor * phase)
        z = Radius * math.sin(phase) * math.exp(-spiralFactor * phase)
        
    elseif CurrentPattern == "Wave" then
        x = Radius * math.cos(phase)
        z = Radius * math.sin(phase * 2)
        
    elseif CurrentPattern == "Infinity" then
        local infinityPhase = phase * 2
        x = Radius * math.cos(infinityPhase) / (1 + math.pow(math.sin(infinityPhase), 2))
        z = Radius * math.sin(infinityPhase) * math.cos(infinityPhase) / (1 + math.pow(math.sin(infinityPhase), 2))
        
    elseif CurrentPattern == "Diamond" then
        local diamondPhase = phase % (math.pi / 2)
        local diamondSide = math.floor(phase / (math.pi / 2)) % 4
        
        if diamondSide == 0 then
            x = Radius * math.cos(diamondPhase)
            z = Radius * math.sin(diamondPhase)
        elseif diamondSide == 1 then
            x = -Radius * math.sin(diamondPhase)
            z = Radius * math.cos(diamondPhase)
        elseif diamondSide == 2 then
            x = -Radius * math.cos(diamondPhase)
            z = -Radius * math.sin(diamondPhase)
        else
            x = Radius * math.sin(diamondPhase)
            z = -Radius * math.cos(diamondPhase)
        end
        
    elseif CurrentPattern == "Cross" then
        local crossPhase = phase % (math.pi / 2)
        local crossSide = math.floor(phase / (math.pi / 2)) % 4
        
        if crossSide == 0 then
            x = Radius
            z = Radius * math.tan(crossPhase)
        elseif crossSide == 1 then
            x = Radius * math.tan(math.pi/2 - crossPhase)
            z = Radius
        elseif crossSide == 2 then
            x = -Radius
            z = -Radius * math.tan(crossPhase)
        else
            x = -Radius * math.tan(math.pi/2 - crossPhase)
            z = -Radius
        end
        
    elseif CurrentPattern == "Arrow" then
        local arrowPhase = phase % (2 * math.pi / 3)
        local arrowSide = math.floor(phase / (2 * math.pi / 3)) % 3
        
        if arrowSide == 0 then
            x = Radius * (1 - arrowPhase / (2 * math.pi / 3))
            z = 0
        elseif arrowSide == 1 then
            local angle = math.pi / 3
            x = Radius * math.cos(angle) * (arrowPhase / (2 * math.pi / 3))
            z = Radius * math.sin(angle) * (arrowPhase / (2 * math.pi / 3))
        else
            local angle = -math.pi / 3
            x = Radius * math.cos(angle) * (arrowPhase / (2 * math.pi / 3))
            z = Radius * math.sin(angle) * (arrowPhase / (2 * math.pi / 3))
        end
        
    elseif CurrentPattern == "Butterfly" then
        local butterflyPhase = phase * 2
        x = Radius * math.sin(butterflyPhase) * (math.exp(math.cos(butterflyPhase)) - 2 * math.cos(4 * butterflyPhase) - math.pow(math.sin(butterflyPhase / 12), 5))
        z = Radius * math.cos(butterflyPhase) * (math.exp(math.cos(butterflyPhase)) - 2 * math.cos(4 * butterflyPhase) - math.pow(math.sin(butterflyPhase / 12), 5))
        x = x * 0.3
        z = z * 0.3
        
    elseif CurrentPattern == "Flower" then
        local petals = 6
        local flowerPhase = phase * petals
        x = Radius * (1 + 0.3 * math.cos(petals * phase)) * math.cos(phase)
        z = Radius * (1 + 0.3 * math.cos(petals * phase)) * math.sin(phase)
    end
    
    return (root.CFrame * CFrame.new(x, y, z)).p
end

local function startGeometricLoop()
    if geometricLoopConn then geometricLoopConn:Disconnect() end
    tAccum = 0
    rescanGeometricTargets()
    if #GeometricList == 0 then 
        IsGeometricActive = false 
        return 
    end

    geometricLoopConn = RunService.Heartbeat:Connect(function(dt)
        local root = LocalHRP(); if not root then return end
        tAccum += dt

        for j = #GeometricList, 1, -1 do
            local rec = GeometricList[j]
            if not rec.part or not rec.part.Parent then
                table.remove(GeometricList, j)
            else
                rec.part.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
                rec.part.AssemblyAngularVelocity = Vector3.new(0, 0, 0)

                rec.bp.Position = getPatternTarget(j, #GeometricList, tAccum)
                rec.bg.CFrame = root.CFrame
            end
        end
    end)
end

local function getWingPosition(i, n, t)
    local root = LocalHRP(); if not root then return Vector3.new(0,0,0) end

    local flapCycle = t * (WingSpeed / 500)
    local side = (i <= math.ceil(n/2)) and 1 or -1
    local wingIndex = (i <= math.ceil(n/2)) and i or (i - math.ceil(n/2))
    local maxWingIndex = math.ceil(n/2)
    local wingProgress = wingIndex / maxWingIndex
    local flapAmount = math.sin(flapCycle + wingProgress * math.pi) * WingSpread

    local x = side * (WingRadius * wingProgress + flapAmount * 0.5)
    local y = WingHeightOffset + math.abs(math.sin(flapCycle + wingProgress * math.pi)) * (WingSpread * 0.5)
    local z = math.sin(flapCycle * 0.5) * (WingRadius * 0.3) + (wingProgress * WingRadius * 0.2)

    return (root.CFrame * CFrame.new(x, y, z)).p
end

local function rescanRotationTargets()
    for _,r in ipairs(RotationList) do 
        if r.part and r.part.Parent then 
            detachPhysics(r.part) 
        end
    end
    RotationList = {}
    partStates = {}
    
    local currentTargets = {}

    for _, d in ipairs(Workspace:GetDescendants()) do
        if d:IsA("Part") and d.Name:match(RotationTargetName) and not d.Anchored then
            table.insert(currentTargets, {model=d.Parent or d, part=d})
        elseif d:IsA("Model") and d.Name:match(RotationTargetName) then
            local part = d.PrimaryPart or d:FindFirstChildWhichIsA("BasePart")
            if part and not part.Anchored then 
                table.insert(currentTargets, {model=d, part=part}) 
            end
        end
    end
    
    if #currentTargets == 0 then return end
    
    for i, target in ipairs(currentTargets) do
        local bp, bg = attachPhysics(target.part)
        
        local startAngle = math.random(0, 360)
        local isRising = math.random() > 0.5
        
        partStates[target.part] = {
            angle = math.rad(startAngle),
            height = MinHeight + (MaxHeight - MinHeight) * math.random(),
            isRising = isRising,
            heightOffset = math.random(0.2, 1.0)
        }
        
        table.insert(RotationList, {
            model = target.model,
            part = target.part,
            id = i,
            bp = bp,
            bg = bg,
            lastPosition = target.part.Position
        })
    end
end

local function ensureSafeDistance(position, rotationIndex, totalRotations)
    local root = LocalHRP()
    if not root then return position end
    
    local playerPos = root.Position
    local toPlayer = playerPos - position
    local distance = toPlayer.Magnitude
    
    if distance < PlayerAvoidDistance then
        local safeDirection = toPlayer.Unit * -1
        local neededDistance = PlayerAvoidDistance - distance
        position = position + safeDirection * (neededDistance + RotationSafetyMargin)
    end
    
    return position
end

local function getBouncingRotationPosition(i, n, t, dt)
    local root = LocalHRP()
    if not root then return Vector3.new(0,0,0) end

    local rec = RotationList[i]
    if not rec or not partStates[rec.part] then return root.Position end
    
    local state = partStates[rec.part]
    
    state.angle = state.angle + math.rad(RotationSpeed2 * dt)
    if state.angle > math.rad(360) then
        state.angle = state.angle - math.rad(360)
    end
    
    if state.isRising then
        state.height = state.height + (LiftSpeed * dt * state.heightOffset)
        if state.height >= MaxHeight then
            state.height = MaxHeight
            state.isRising = false
        end
    else
        state.height = state.height - (LowerSpeed * dt * state.heightOffset)
        if state.height <= MinHeight then
            state.height = MinHeight
            state.isRising = true
        end
    end
    
    local x = math.cos(state.angle) * RotationRadius
    local z = math.sin(state.angle) * RotationRadius
    local y = state.height
    
    local position = (root.CFrame * CFrame.new(x, y, z)).p
    
    return ensureSafeDistance(position, i, n)
end

local function getRotationOrientation(partPosition, root, state)
    local toCenter = (root.Position - partPosition).Unit
    local upVector = Vector3.new(0, 1, 0)
    
    local baseCF = CFrame.lookAt(partPosition, partPosition + toCenter)
    
    local tiltCF = baseCF * CFrame.fromEulerAnglesXYZ(
        math.rad(15 * math.sin(state.angle)),
        math.rad(10 * math.cos(state.angle)),
        0
    )
    
    return tiltCF
end

local function startRotationLoop()
    if rotationLoopConn then rotationLoopConn:Disconnect() end
    rescanRotationTargets()
    if #RotationList == 0 then 
        IsRotationActive = false 
        return 
    end

    rotationLoopConn = RunService.Heartbeat:Connect(function(dt)
        local root = LocalHRP()
        if not root then return end

        for j = #RotationList, 1, -1 do
            local rec = RotationList[j]
            if not rec.part or not rec.part.Parent then
                partStates[rec.part] = nil
                table.remove(RotationList, j)
            else
                local targetPos = getBouncingRotationPosition(j, #RotationList, 0, dt)
                
                if rec.lastPosition then
                    local lerpFactor = 0.3
                    local smoothPos = rec.lastPosition:Lerp(targetPos, lerpFactor)
                    rec.bp.Position = smoothPos
                    rec.lastPosition = smoothPos
                else
                    rec.bp.Position = targetPos
                    rec.lastPosition = targetPos
                end
                
                if partStates[rec.part] then
                    local targetCF = getRotationOrientation(targetPos, root, partStates[rec.part])
                    rec.bg.CFrame = targetCF
                end
            end
        end
    end)
end

-- ====================================================================
local function GetPlayersInOrder()
    local allPlayers = {}
    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LP and player.Character then
            table.insert(allPlayers, player)
        end
    end
    
    -- 現在位置を基準にソート（近い順に）
    local myHRP = LocalHRP()
    if myHRP then
        table.sort(allPlayers, function(a, b)
            local charA = a.Character
            local charB = b.Character
            local hrpA = charA and charA:FindFirstChild("HumanoidRootPart")
            local hrpB = charB and charB:FindFirstChild("HumanoidRootPart")
            
            if hrpA and hrpB then
                local distA = (myHRP.Position - hrpA.Position).Magnitude
                local distB = (myHRP.Position - hrpB.Position).Magnitude
                return distA < distB
            end
            return true
        end)
    end
    
    return allPlayers
end

local function TeleportToPlayer(player)
    if not player or player == LP then return end
    local targetChar = player.Character
    if not targetChar then return end
    local targetHRP = targetChar:FindFirstChild("HumanoidRootPart")
    if not targetHRP then return end
    
    local myHRP = LocalHRP()
    if not myHRP then return end
    
    -- 横（隣）にテレポート（カメラの向きに5ユニット横）
    local camera = Workspace.CurrentCamera
    local cameraCF = camera.CFrame
    local rightVector = cameraCF.RightVector
    local teleportPosition = targetHRP.Position + (rightVector * 5)
    
    myHRP.CFrame = CFrame.new(teleportPosition, targetHRP.Position)
    print("テレポート先: " .. player.Name)
end

local function StartAllTeleportLoop()
    if AllTeleportActive then return end
    AllTeleportActive = true
    AllTeleportIndex = 1
    
    local allPlayers = GetPlayersInOrder()
    
    if #allPlayers == 0 then
        AllTeleportActive = false
        return
    end
    
    task.spawn(function()
        while AllTeleportActive do
            local player = allPlayers[AllTeleportIndex]
            if player then
                TeleportToPlayer(player)
            end
            
            AllTeleportIndex = AllTeleportIndex + 1
            if AllTeleportIndex > #allPlayers then
                AllTeleportIndex = 1
            end
            
            task.wait(AllTeleportInterval)
        end
    end)
end

local function StopAllTeleportLoop()
    AllTeleportActive = false
end

local function GetNearPartsInSphere(origin, radius, height)
    local sphere = Instance.new("Part")
    sphere.Size = Vector3.new(radius * 2, height * 2, radius * 2)
    sphere.Position = origin
    sphere.Anchored = true
    sphere.CanCollide = false
    sphere.Transparency = 1
    sphere.Parent = Workspace
    
    local parts = Workspace:GetPartsInPart(sphere)
    sphere:Destroy()
    
    local partsInSphere = {}
    for _, part in ipairs(parts) do
        local distance = (part.Position - origin).Magnitude
        local heightDiff = math.abs(part.Position.Y - origin.Y)
        
        if distance <= radius and heightDiff <= height / 2 then
            table.insert(partsInSphere, part)
        end
    end
    
    return partsInSphere
end

-- オーラループ
-- ====================================================================
RunService.Heartbeat:Connect(function(dt)
    AuraTimer = AuraTimer + dt
    local root = LocalHRP()
    if not root then return end
    
    if Config.Target.TeleportToTarget and Config.Target.SelectedPlayer then
        local targetChar = Config.Target.SelectedPlayer.Character
        if targetChar and targetChar:FindFirstChild("HumanoidRootPart") then
            root.CFrame = targetChar.HumanoidRootPart.CFrame + Vector3.new(0, 5, 0)
        end
    end
    
    if AuraTimer >= 0.5 then
        local anyAuraActive = false
        for _, v in pairs(Config.Aura) do
            if type(v) == "boolean" and v then
                anyAuraActive = true
                break
            end
        end
        
        if anyAuraActive then
            local nearParts = GetNearPartsInSphere(root.Position, Config.Aura.Radius, Config.Aura.Height)
            
            for _, part in ipairs(nearParts) do
                if part.Name == "HumanoidRootPart" and not part:IsDescendantOf(LP.Character) then
                    local character = part.Parent
                    SetNetworkOwner(part)
                    
                    affectedPlayers[character] = true
                    
                    if Config.Aura.SlowLegsAura then MakeSlowLegs(character) end
                    if Config.Aura.IrritateAura then IrritatePlayer(character) end
                    
                    if Config.Aura.BlastAura then 
                        local direction = (part.Position - root.Position).Unit
                        Velocity(part, direction * Config.Aura.BlastPower)
                    end
                    if Config.Aura.SpinAura then 
                        local bav = Instance.new("BodyAngularVelocity")
                        bav.MaxTorque = Vector3.new(1e8, 1e8, 1e8)
                        bav.AngularVelocity = Vector3.new(0, Config.Aura.SpinSpeed, 0)
                        bav.Parent = part
                        Debris:AddItem(bav, 0.1)
                    end
                    if Config.Aura.FlingAura then 
                        flingMemory[character] = true
                        local randomDir = Vector3.new(
                            math.random(-100, 100),
                            math.random(50, 100),
                            math.random(-100, 100)
                        ).Unit
                        Velocity(part, randomDir * Config.Aura.FlingPower)
                    end
                    if Config.Aura.KillAura then 
                        MoveTo(part, CFrame.new(4096, -75, 4096)) 
                        Velocity(part, Vector3.new(0, -1000, 0)) 
                    end
                    if Config.Aura.VoidAura then Velocity(part, Vector3.new(0, 10000, 0)) end
                end
            end
        end
        AuraTimer = 0
    end
    
    for character, _ in pairs(affectedPlayers) do
        if character and character.Parent then
            if Config.Aura.SlowLegsAura then
                MakeSlowLegs(character)
            end
            
            if Config.Aura.IrritateAura then
                IrritatePlayer(character)
            end
            
            if flingMemory[character] and Config.Aura.FlingAura then
                local root = character:FindFirstChild("HumanoidRootPart")
                if root then
                    local randomDir = Vector3.new(
                        math.random(-100, 100),
                        math.random(50, 100),
                        math.random(-100, 100)
                    ).Unit
                    Velocity(root, randomDir * Config.Aura.FlingPower / 2)
                end
            end
        else
            affectedPlayers[character] = nil
            flingMemory[character] = nil
        end
    end
    
    if Config.Target.KillAllLoop then
        task.spawn(function()
            for _, player in ipairs(Players:GetPlayers()) do
                if player ~= LP and player.Character then
                    KillPlayer(player)
                    task.wait(0.5)
                end
            end
        end)
    end
end)
