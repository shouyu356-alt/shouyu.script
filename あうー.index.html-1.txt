        // ===== ユーティリティ関数 =====
        function generateRandomString(length) {
            const chars = 'abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ';
            let result = '';
            for (let i = 0; i < length; i++) {
                result += chars.charAt(Math.floor(Math.random() * chars.length));
            }
            return result;
        }

        function generateRandomVarName() {
            const prefixes = ['_', '__', '___'];
            const prefix = prefixes[Math.floor(Math.random() * prefixes.length)];
            return prefix + generateRandomString(Math.floor(Math.random() * 20) + 10);
        }

        function encryptString(str) {
            const key = Math.floor(Math.random() * 255) + 1;
            let encrypted = [];
            for (let i = 0; i < str.length; i++) {
                encrypted.push(str.charCodeAt(i) ^ key);
            }
            return { encrypted: encrypted, key: key };
        }

        // ===== Toast通知 =====
        let toastTimeout = null;

        function showToast(message, duration = 2500) {
            const toast = document.getElementById('toast');
            toast.textContent = message;
            toast.classList.add('show');
            
            clearTimeout(toastTimeout);
            toastTimeout = setTimeout(() => {
                toast.classList.remove('show');
            }, duration);
        }

        // ===== 難読化エンジン =====
        function createVMEngine() {
            const vmName = generateRandomVarName();
            const decoderName = generateRandomVarName();
            const keyName = generateRandomVarName();
            
            return `
local ${vmName} = {}
${vmName}.${keyName} = function(${generateRandomVarName()}, ${generateRandomVarName()})
    local ${generateRandomVarName()} = {}
    for ${generateRandomVarName()} = 1, #${generateRandomVarName()} do
        ${generateRandomVarName()}[${generateRandomVarName()}] = string.char(bit32.bxor(${generateRandomVarName()}[${generateRandomVarName()}], ${generateRandomVarName()}))
    end
    return table.concat(${generateRandomVarName()})
end

${vmName}.${decoderName} = function(${generateRandomVarName()})
    local ${generateRandomVarName()} = {}
    for ${generateRandomVarName()} in string.gmatch(${generateRandomVarName()}, "[^,]+") do
        table.insert(${generateRandomVarName()}, tonumber(${generateRandomVarName()}))
    end
    return ${generateRandomVarName()}
end
`;
        }

        function createAntiDebug() {
            const varName = generateRandomVarName();
            const funcName = generateRandomVarName();
            return `
local ${varName} = tick()
local function ${funcName}()
    if tick() - ${varName} > 0.5 then
        while true do end
    end
    ${varName} = tick()
end
task.spawn(function()
    while task.wait(0.1) do
        ${funcName}()
    end
end)
`;
        }

        function createDummyCode(lines) {
            let dummy = '';
            for (let i = 0; i < lines; i++) {
                const varName = generateRandomVarName();
                const operations = [
                    `local ${varName} = ${Math.floor(Math.random() * 1000)}`,
                    `local ${varName} = "${generateRandomString(20)}"`,
                    `local ${varName} = function() return ${Math.random()} end`,
                    `local ${varName} = {${Math.random()}, ${Math.random()}, ${Math.random()}}`,
                ];
                dummy += operations[Math.floor(Math.random() * operations.length)] + '\n';
            }
            return dummy;
        }

        function wrapInControlFlow(code) {
            const stateName = generateRandomVarName();
            const switchName = generateRandomVarName();
            const lines = code.split('\n').filter(line => line.trim());
            
            if (lines.length === 0) return code;
            
            let wrapped = `local ${stateName} = 1\n`;
            wrapped += `local ${switchName} = {}\n`;
            
            lines.forEach((line, index) => {
                const nextState = index + 2;
                wrapped += `${switchName}[${index + 1}] = function()\n`;
                wrapped += `    ${line}\n`;
                wrapped += `    ${stateName} = ${nextState}\n`;
                wrapped += `end\n`;
            });
            
            wrapped += `while ${stateName} <= ${lines.length} do\n`;
            wrapped += `    if ${switchName}[${stateName}] then\n`;
            wrapped += `        ${switchName}[${stateName}]()\n`;
            wrapped += `    else\n`;
            wrapped += `        break\n`;
            wrapped += `    end\n`;
            wrapped += `end\n`;
            
            return wrapped;
        }

        function obfuscateVariables(code) {
            const varPattern = /local\s+([a-zA-Z_][a-zA-Z0-9_]*)/g;
            const varMap = new Map();
            let match;
            
            while ((match = varPattern.exec(code)) !== null) {
                const originalName = match[1];
                if (!varMap.has(originalName)) {
                    varMap.set(originalName, generateRandomVarName());
                }
            }
            
            varMap.forEach((newName, oldName) => {
                const regex = new RegExp(`\\b${oldName}\\b`, 'g');
                code = code.replace(regex, newName);
            });
            
            return code;
        }

        function expandToTargetSize(code, targetSize) {
            let currentSize = code.length;
            let expanded = code;
            
            let iterations = 0;
            while (currentSize < targetSize && iterations < 50) {
                iterations++;
                const neededSize = targetSize - currentSize;
                const dummyLines = Math.min(Math.ceil(neededSize / 80), 300);
                const dummyCode = createDummyCode(dummyLines);
                
                const insertPosition = Math.floor(Math.random() * (expanded.length + 1));
                expanded = expanded.slice(0, insertPosition) + '\n' + dummyCode + '\n' + expanded.slice(insertPosition);
                
                currentSize = expanded.length;
            }
            
            return expanded;
        }

        // ===== メイン難読化処理 =====
        async function obfuscateCode() {
            const startTime = performance.now();
            const inputCode = document.getElementById('inputCode').value.trim();
            
            if (!inputCode) {
                showToast('⚠️ スクリプトを入力してください', 2000);
                return;
            }

            const progress = document.getElementById('progress');
            const progressFill = document.getElementById('progressFill');
            progress.style.display = 'block';

            const options = {
                stringEncryption: document.getElementById('stringEncryption').checked,
                vmProtection: document.getElementById('vmProtection').checked,
                controlFlow: document.getElementById('controlFlow').checked,
                antiDebug: document.getElementById('antiDebug').checked,
                constantWrapping: document.getElementById('constantWrapping').checked,
                deadCode: document.getElementById('deadCode').checked,
            };

            let obfuscated = '';

            // Stage 1: VM Engine
            progressFill.style.width = '10%';
            progressFill.textContent = '10%';
            await sleep(80);
            
            if (options.vmProtection) {
                obfuscated += createVMEngine() + '\n\n';
            }

            // Stage 2: Anti-Debug
            progressFill.style.width = '20%';
            progressFill.textContent = '20%';
            await sleep(80);
            
            if (options.antiDebug) {
                obfuscated += createAntiDebug() + '\n\n';
            }

            // Stage 3: Variable Obfuscation
            progressFill.style.width = '40%';
            progressFill.textContent = '40%';
            await sleep(80);
            
            let processedCode = obfuscateVariables(inputCode);

            // Stage 4: String Encryption
            progressFill.style.width = '60%';
            progressFill.textContent = '60%';
            await sleep(80);
            
            if (options.stringEncryption) {
                processedCode = processedCode.replace(/(["'])(?:(?=(\\?))\2.)*?\1/g, (match) => {
                    const str = match.slice(1, -1);
                    const { encrypted, key } = encryptString(str);
                    return `loadstring(string.char(${encrypted.map(c => c ^ key).join(',')}))()`;
                });
            }

            // Stage 5: Control Flow
            progressFill.style.width = '80%';
            progressFill.textContent = '80%';
            await sleep(80);
            
            if (options.controlFlow) {
                processedCode = wrapInControlFlow(processedCode);
            }

            obfuscated += processedCode;

            // Stage 6: Expand to target size
            progressFill.style.width = '90%';
            progressFill.textContent = '90%';
            await sleep(80);
            
            const targetSize = 100000 + Math.floor(Math.random() * 50000);
            obfuscated = expandToTargetSize(obfuscated, targetSize);

            // Stage 7: Final dummy code
            if (options.deadCode) {
                obfuscated = createDummyCode(300) + '\n' + obfuscated + '\n' + createDummyCode(300);
            }

            // Complete
            progressFill.style.width = '100%';
            progressFill.textContent = '100%';
            
            const endTime = performance.now();
            const processTime = Math.round(endTime - startTime);

            document.getElementById('outputCode').value = obfuscated;
            document.getElementById('originalChars').textContent = inputCode.length.toLocaleString();
            document.getElementById('obfuscatedChars').textContent = obfuscated.length.toLocaleString();
            document.getElementById('expansionRatio').textContent = Math.round(obfuscated.length / Math.max(inputCode.length, 1)) + 'x';
            document.getElementById('processTime').textContent = processTime + 'ms';

            showToast(`✅ 難読化完了！ ${obfuscated.length.toLocaleString()}文字`, 2000);

            setTimeout(() => {
                progress.style.display = 'none';
                progressFill.style.width = '0%';
            }, 800);
        }

        function sleep(ms) {
            return new Promise(resolve => setTimeout(resolve, ms));
        }

        // ===== UI操作 =====
        function copyOutput() {
            const outputCode = document.getElementById('outputCode');
            if (!outputCode.value) {
                showToast('⚠️ コピーするコードがありません', 2000);
                return;
            }
            
            if (navigator.clipboard && navigator.clipboard.writeText) {
                navigator.clipboard.writeText(outputCode.value).then(() => {
                    showToast('📋 クリップボードにコピーしました！', 2000);
                }).catch(() => {
                    fallbackCopy(outputCode);
                });
            } else {
                fallbackCopy(outputCode);
            }
        }

        function fallbackCopy(textarea) {
            textarea.select();
            try {
                document.execCommand('copy');
                showToast('📋 コピーしました！', 2000);
            } catch (e) {
                showToast('⚠️ コピーに失敗しました', 2000);
            }
            textarea.blur();
        }

        function downloadTxt() {
            const outputCode = document.getElementById('outputCode').value;
            if (!outputCode) {
                showToast('⚠️ ダウンロードするコードがありません', 2000);
                return;
            }
            
            try {
                const blob = new Blob([outputCode], { type: 'text/plain;charset=utf-8' });
                const url = window.URL.createObjectURL(blob);
                const link = document.createElement('a');
                link.href = url;
                
                const timestamp = new Date().toISOString().replace(/[:.]/g, '-').slice(0, -5);
                link.download = `obfuscated_script_${timestamp}.txt`;
                
                document.body.appendChild(link);
                link.click();
                document.body.removeChild(link);
                window.URL.revokeObjectURL(url);
